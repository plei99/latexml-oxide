\begin{algorithm}[H]
\caption{Euclid}
\begin{algorithmic}[1]
  \Function{gcd}{$a, b$}
    \While{$b \neq 0$}
      \State $(a, b) \gets (b, a \bmod b)$
    \EndWhile
    \State \Return $a$
  \EndFunction
\end{algorithmic}
\end{algorithm}
